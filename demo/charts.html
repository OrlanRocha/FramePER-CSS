HTML5 Nativo 
<div class="frame-chart"
     data-chart="area"
     data-chart-options='{"height":220,"format":"currency"}'
     data-chart-data='{"labels":["Jan","Fev",...],"series":[{"name":"Faturamento","data":[18000,...]}]}'>
</div>

Guia de Uso & API Completa

Como instanciar e manipular instâncias de gráficos programaticamente

// 1. Instanciação rápida
const chart = new FramePER.Chart('#meu-grafico', {
  type: 'line', // 'line' | 'area' | 'bar' | 'horizontal-bar' | 'mixed'
                // 'donut' | 'pie' | 'gauge' | 'radar' | 'sparkline'
  data: {
    labels: ['Jan', 'Fev', 'Mar', 'Abr', 'Mai', 'Jun'],
    series: [
      { name: 'Vendas', data: [120, 190, 160, 240, 210, 310], color: 'var(--primary)' }
      // No tipo 'mixed', defina type: 'bar' ou type: 'line' por série!
    ]
  },
  options: {
    height: 280,
    curved: true,
    format: 'currency', // 'currency' (R$), 'percent' (%), 'number'
    tooltip: true,
    legend: true
  }
});

// 2. Atualização reativa de dados (com animação)
chart.update({
  series: [{ name: 'Vendas', data: [150, 230, 190, 280, 260, 390] }]
});

// 3. Exportar como imagem
chart.exportPNG('relatorio-vendas.png');
